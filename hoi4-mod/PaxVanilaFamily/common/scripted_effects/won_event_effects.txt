			check_variable = { won_inflation > 25 }

		}
		set_country_flag = { flag = won_hyper_cd days = 365 }
		country_event = { id = won.37 }

		limit = {
			check_variable = { won_debt_ratio > 1 }

		}

			limit = { has_game_rule = { rule = won_event_rate option = WON_EVENTS_LOW } }
			set_temp_variable = { won_chance = 15 }
		}
		set_temp_variable_to_random = { var = won_roll min = 0 max = 100 }
		if = {
			limit = {
				check_variable = { won_roll < won_chance }

				10 = {	# Stock market rally

				}
				10 = {	# Market crash
					modifier = { factor = 0.25 has_country_flag = won_prog_bank }
					modifier = { factor = 0 check_variable = { won_confidence > 45 } NOT = { check_variable = { won_system = 1 } } }
					modifier = { factor = 0 check_variable = { won_system = 4 } }
