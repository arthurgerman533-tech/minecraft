	add_to_variable = { won_conf_shock = -6 }

# The regime changed and the old system is no longer legal.
won_system_lost = {
	won_remove_system_ideas = yes
	won_clear_system_vars = yes
	set_variable = { won_system = 0 }
	set_variable = { won_system_cd = 0 }
	if = {
		limit = { is_ai = yes }
		won_choose_default_system = yes
	}
	else = {
		country_event = { id = won.3 }
	}
}

won_cancel_all_trade = {
	for_each_loop = {
		array = won_trade
		value = won_p
		var:won_p = {
			clr_country_flag = won_trade_@PREV
			remove_from_array = { won_trade = PREV }
			PREV = { clr_country_flag = won_trade_@PREV }
		}
	}
	clear_array = won_trade
	set_variable = { won_trade_count = 0 }
}
