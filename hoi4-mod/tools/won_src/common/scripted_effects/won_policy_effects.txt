# Wealth of Nations - dashboard policy buttons. Each lever has a 30-day cooldown.
won_policy_refresh = {
	won_recalc_modifiers = yes
	won_update_charts = yes
}

won_tax_down = {
	subtract_from_variable = { won_tax = 1 }
	clamp_variable = { var = won_tax min = 1 max = 3 }
	clamp_variable = { var = won_tax_low max = won_tax }
	set_country_flag = { flag = won_tax_cd days = 30 }
	won_policy_refresh = yes
}
won_tax_up = {
	add_to_variable = { won_tax = 1 }
	clamp_variable = { var = won_tax min = 1 max = 3 }
	set_country_flag = { flag = won_tax_cd days = 30 }
	won_policy_refresh = yes
}
won_rate_down = {
	subtract_from_variable = { won_rate = 1 }
	clamp_variable = { var = won_rate min = 1 max = 3 }
	clamp_variable = { var = won_rate_low max = won_rate }
	set_country_flag = { flag = won_rate_cd days = 30 }
	won_policy_refresh = yes
}
won_rate_up = {
	add_to_variable = { won_rate = 1 }
	clamp_variable = { var = won_rate min = 1 max = 3 }
	clamp_variable = { var = won_rate_high min = won_rate }
	set_country_flag = { flag = won_rate_cd days = 30 }
	won_policy_refresh = yes
}
won_budget_down = {
	set_country_flag = won_budget_used
	subtract_from_variable = { won_budget = 1 }
	if = {
		limit = { check_variable = { won_budget < 0 } }
		set_variable = { won_budget = 4 }
	}
	set_country_flag = { flag = won_budget_cd days = 30 }
	won_policy_refresh = yes
}
won_budget_up = {
	set_country_flag = won_budget_used
	add_to_variable = { won_budget = 1 }
	if = {
		limit = { check_variable = { won_budget > 4 } }
		set_variable = { won_budget = 0 }
	}
	set_country_flag = { flag = won_budget_cd days = 30 }
	won_policy_refresh = yes
}
won_open_toggle = {
	if = {
		limit = { check_variable = { won_open = 1 } }
		set_variable = { won_open = 0 }
	}
	else = {
		set_variable = { won_open = 1 }
	}
	set_country_flag = { flag = won_open_cd days = 30 }
	won_policy_refresh = yes
}

	country_event = { id = won.2 }
}
