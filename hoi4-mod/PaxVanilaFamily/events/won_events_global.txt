######################################################################
# The Foreign Trade Office: one place for everything that comes from abroad.
######################################################################
country_event = {
	id = won.220
	title = won.220.t
	desc = won.220.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes
	immediate = { won_office_refresh = yes }
	option = {	# Next request
		name = won.220.a
		trigger = { check_variable = { won_q_count > 0 } }
		won_q_open_next = yes
	}
	option = {	# Sell surplus for cash
		name = won.220.b
		trigger = {
			has_country_flag = won_arms_agency
			won_has_arms_surplus = yes
			NOT = { has_country_flag = won_dump_cd }
		}
		country_event = { id = won.205 }
	}
	option = {	# International arms exhibition (Regional Arms Supplier or better)
		name = won.220.c
		trigger = {
			has_country_flag = won_arms_agency
			check_variable = { aon_tier > 1 }
			has_war = no
			NOT = { has_country_flag = won_fair_cd }
			check_variable = { won_treasury > won_c_small }
		}
		set_temp_variable = { won_spend = won_c_small }
		won_pay = yes
		custom_effect_tooltip = won_tt_pay_small
		aon_hold_exhibition_effect = yes
		set_country_flag = { flag = won_fair_cd days = 365 }
		country_event = { id = aon.10 days = 30 }
		won_office_return = yes
	}
	option = {	# Export credit guarantees (Established Arms Exporter or better)
		name = won.220.e
		trigger = {
			has_country_flag = won_arms_agency
			check_variable = { aon_tier > 2 }
			NOT = { has_country_flag = won_credit_cd }
			check_variable = { won_treasury > won_c_mid }
		}
		set_temp_variable = { won_spend = won_c_mid }
		won_pay = yes
		custom_effect_tooltip = won_tt_pay_mid
		aon_export_credit_effect = yes
		set_country_flag = { flag = won_credit_cd days = 365 }
		won_office_return = yes
	}
	option = {	# Study imported equipment
		name = won.220.s
		trigger = {
			check_variable = { aon_insight > 39.99 }
			aon_has_studiable_imports_trigger = yes
			NOT = { has_country_flag = won_study_cd }
		}
		subtract_from_variable = { aon_insight = 40 }
		set_country_flag = { flag = won_study_cd days = 90 }
		custom_effect_tooltip = won_tt_study
		country_event = { id = aon.20 }
	}
	option = {	# Start courting foreign buyers
		name = won.220.f
		trigger = {
			has_country_flag = won_arms_agency
			NOT = { has_country_flag = won_court_buyers }
		}
		set_country_flag = won_court_buyers
		custom_effect_tooltip = won_tt_court_on
		won_office_return = yes
	}
	option = {	# Stop courting foreign buyers
		name = won.220.g
		trigger = { has_country_flag = won_court_buyers }
		clr_country_flag = won_court_buyers
		custom_effect_tooltip = won_tt_court_off
		won_office_return = yes
	}
	option = {	# Switch to pop-ups
		name = won.220.h
		trigger = { has_country_flag = won_q_manual }
		clr_country_flag = won_q_manual
		won_office_return = yes
	}
	option = {	# Switch to the in-tray
		name = won.220.i
		trigger = { NOT = { has_country_flag = won_q_manual } }
		set_country_flag = won_q_manual
		won_office_return = yes
	}
	option = {	# Close
		name = won.220.x
	}
}

			check_variable = { won_sc_count < won_sc_max }

		ai_chance = { factor = 20 }

	}
