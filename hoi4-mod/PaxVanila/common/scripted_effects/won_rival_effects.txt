# Wealth of Nations - arms customers, market rivals and embargoes.
# Country scope, ROOT = THIS = the human exporter unless noted.

###########################################################################
# Customers (monthly, human players). Only countries on aon_clients are told to prefer
# our listings on the International Market. How many depends on our Market Standing:
# 3 at the start, +3 per tier, 18 as Arsenal of the World. Close partners are picked first.
###########################################################################
aon_pick_clients = {
	clear_array = aon_clients
	set_variable = { aon_client_n = 0 }
	set_variable = { aon_client_cap = aon_tier }
	multiply_variable = { aon_client_cap = 3 }
	add_to_variable = { aon_client_cap = 3 }
	if = {
		limit = { NOT = { has_country_flag = won_court_buyers } }
		set_variable = { aon_client_cap = 0 }
	}
	every_other_country = {
		limit = { aon_client_base = yes aon_client_close = yes }
		if = {
			limit = { ROOT = { check_variable = { aon_client_n < aon_client_cap } } }
			ROOT = {
				add_to_array = { aon_clients = PREV }
				add_to_variable = { aon_client_n = 1 }
			}
		}
	}
	every_other_country = {
		limit = { aon_client_base = yes aon_client_close = no aon_client_open = yes }
		if = {
			limit = { ROOT = { check_variable = { aon_client_n < aon_client_cap } } }
			ROOT = {
				add_to_array = { aon_clients = PREV }
				add_to_variable = { aon_client_n = 1 }
			}
		}
	}
}

###########################################################################
# Market rival (monthly, human players): the AI great power outside our faction
# with the most arms factories. It competes for our customers.
###########################################################################
won_find_rival = {
	clear_variable = won_rival
	set_temp_variable = { won_best = 0 }
	every_other_country = {
		limit = {
			is_ai = yes
			is_major = yes
			exists = yes
			has_capitulated = no
			NOT = { is_in_faction_with = ROOT }
			NOT = { is_subject_of = ROOT }
		}
		set_temp_variable = { won_t = num_of_military_factories }
		if = {
			limit = { check_variable = { var = won_t value = won_best compare = greater_than } }
			set_temp_variable = { won_best = won_t }
			ROOT = { set_variable = { won_rival = PREV } }
		}
	}
	if = {
		limit = { has_variable = won_rival }
		var:won_rival = { set_variable = { global.won_rival_name = THIS } }
	}
}

# Monthly rivalry: the rival tries to steal a customer, or one of theirs wants to switch to us.
won_rival_tick = {
	won_find_rival = yes
	if = {
		limit = {
			has_variable = won_rival
			has_country_flag = won_arms_agency
			has_country_flag = won_court_buyers
			NOT = { has_country_flag = won_rival_cd }
			NOT = { has_country_flag = won_q_busy }
		}
		set_temp_variable_to_random = { var = won_roll min = 0 max = 100 }
		if = {
			limit = {
				check_variable = { won_roll < 7 }
				check_variable = { aon_client_n > 1 }
			}
			random_scope_in_array = {
				array = aon_clients
				limit = { exists = yes NOT = { is_in_faction_with = ROOT } }
				ROOT = {
					set_variable = { won_rival_target = PREV }
					set_country_flag = { flag = won_rival_cd days = 150 }
					country_event = { id = won.230 days = 1 }
				}
			}
		}
		else_if = {
			limit = {
				check_variable = { won_roll < 12 }
				check_variable = { aon_tier > 0 }
				check_variable = { aon_client_n < aon_client_cap }
			}
			random_other_country = {
				limit = {
					aon_client_base = yes
					NOT = { is_in_faction_with = ROOT }
					ROOT = { NOT = { is_in_array = { array = aon_clients value = PREV } } }
				}
				ROOT = {
					set_variable = { won_rival_target = PREV }
					set_country_flag = { flag = won_rival_cd days = 150 }
					country_event = { id = won.231 days = 1 }
				}
			}
		}
	}
}

# Event helper: names for the rival and the contested customer.
won_rival_names = {
	if = {
		limit = { has_variable = won_rival }
		var:won_rival = { set_variable = { global.won_rival_name = THIS } }
	}
	if = {
		limit = { has_variable = won_rival_target }
		var:won_rival_target = { set_variable = { global.won_rt_name = THIS } }
	}
}

# Losing a customer to the rival: they will not buy from us for a year.
won_lose_client = {
	var:won_rival_target = {
		set_country_flag = { flag = aon_lost_@ROOT days = 365 }
	}
	remove_from_array = { array = aon_clients value = won_rival_target }
	subtract_from_variable = { aon_client_n = 1 }
	clamp_variable = { var = aon_client_n min = 0 }
}

# Price of matching the rival's offer: 0.15 bn + 0.05 bn per standing tier.
won_rival_match_cost = {
	set_variable = { won_rival_cost = aon_tier }
	multiply_variable = { won_rival_cost = 0.05 }
	add_to_variable = { won_rival_cost = 0.15 }
}

###########################################################################
# Embargoes. THIS = target, ROOT = the country imposing it (diplomatic action scope).
###########################################################################
won_impose_embargo = {
	if = {
		limit = { OR = { won_trade_with_root = yes won_pact_with_root = yes } }
		won_break_agreements = yes
	}
	if = {
		limit = { has_country_flag = won_sc_from_@ROOT }
		set_country_flag = { flag = won_sc_cancel_@ROOT days = 45 }
		clr_country_flag = won_sc_from_@ROOT
	}
	ROOT = { set_country_flag = won_embargo_on_@PREV }
	add_to_array = { won_emb_by = ROOT }
	add_opinion_modifier = { target = ROOT modifier = won_embargoed }
	if = {
		limit = { is_ai = no }
		ROOT = { set_variable = { global.won_emb_name = THIS } }
		country_event = { id = won.232 }
	}
}

won_lift_embargo = {
	ROOT = { clr_country_flag = won_embargo_on_@PREV }
	remove_from_array = { array = won_emb_by value = ROOT }
	remove_opinion_modifier = { target = ROOT modifier = won_embargoed }
}
