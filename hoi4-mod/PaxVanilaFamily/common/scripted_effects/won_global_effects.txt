# Investment climate 0..100 (monthly, every country, after indicators)
###########################################################################
won_update_climate = {
	set_temp_variable = { won_cl = 30 }
	set_temp_variable = { won_t = won_confidence }
	subtract_from_temp_variable = { won_t = 50 }
	multiply_temp_variable = { won_t = 0.5 }
	add_to_temp_variable = { won_cl = won_t }
	set_temp_variable = { won_t = won_growth_y }
	clamp_temp_variable = { var = won_t min = -7.5 max = 7.5 }
	multiply_temp_variable = { won_t = 2 }
	add_to_temp_variable = { won_cl = won_t }
	set_temp_variable = { won_t = stability }
	subtract_from_temp_variable = { won_t = 0.5 }
	multiply_temp_variable = { won_t = 30 }
	add_to_temp_variable = { won_cl = won_t }
	set_temp_variable = { won_t = won_trade_weight }
	multiply_temp_variable = { won_t = 2 }
	add_to_temp_variable = { won_cl = won_t }
	if = { limit = { check_variable = { won_credit = 1 } } add_to_temp_variable = { won_cl = 15 } }
	if = { limit = { check_variable = { won_credit = 2 } } add_to_temp_variable = { won_cl = 10 } }
	if = { limit = { check_variable = { won_credit = 3 } } add_to_temp_variable = { won_cl = 5 } }
	if = { limit = { check_variable = { won_credit = 5 } } subtract_from_temp_variable = { won_cl = 5 } }
	if = { limit = { check_variable = { won_credit = 6 } } subtract_from_temp_variable = { won_cl = 10 } }
	if = { limit = { check_variable = { won_credit = 7 } } subtract_from_temp_variable = { won_cl = 20 } }
	if = { limit = { has_war = yes } subtract_from_temp_variable = { won_cl = 15 } }
	# Vanilla trade and mobilisation laws (won_law_effects.txt)
	add_to_temp_variable = { won_cl = won_law_cl }
	# A place in the World Economic League (won_rank_effects.txt)
	won_rank_climate = yes
	add_to_temp_variable = { won_cl = won_rank_cl }
	add_to_temp_variable = { won_cl = won_sph_cl }
	if = { limit = { has_country_flag = won_prog_bank } add_to_temp_variable = { won_cl = 5 } }
	if = { limit = { check_variable = { won_system = 1 } } add_to_temp_variable = { won_cl = 10 } }
	if = { limit = { check_variable = { won_system = 5 } } add_to_temp_variable = { won_cl = 5 } }
	if = { limit = { check_variable = { won_system = 12 } } add_to_temp_variable = { won_cl = 5 } }
	if = { limit = { check_variable = { won_system = 10 } } subtract_from_temp_variable = { won_cl = 5 } }
	if = { limit = { has_country_flag = won_roadshow } add_to_temp_variable = { won_cl = 15 } }
	if = { limit = { has_country_flag = won_tax_holiday } add_to_temp_variable = { won_cl = 20 } }
	if = { limit = { has_country_flag = won_red_tape } add_to_temp_variable = { won_cl = 10 } }
	if = { limit = { has_country_flag = won_sez } add_to_temp_variable = { won_cl = 10 } }
	if = { limit = { has_idea = won_nationalized } subtract_from_temp_variable = { won_cl = 15 } }
	if = { limit = { has_idea = won_default } subtract_from_temp_variable = { won_cl = 20 } }
	if = {
		limit = {
			OR = {
				check_variable = { won_open = 0 }
				won_closed_to_capital = yes
			}
		}
		set_temp_variable = { won_cl = 0 }
	}
	clamp_temp_variable = { var = won_cl min = 0 max = 100 }
	set_variable = { won_climate = won_cl }
	# Monthly chance (percent) that a foreign investor knocks: a quarter of the climate.
	set_variable = { won_fdi_chance = won_cl }
	multiply_variable = { won_fdi_chance = 0.25 }
}

###########################################################################
# Foreign investors arrive (monthly, every country). ROOT = THIS = host.
###########################################################################
won_pick_investor = {
	set_temp_variable = { won_found = 0 }
	set_temp_variable_to_random = { var = won_r min = 0 max = 100 }
	if = {
		limit = {
			check_variable = { won_r < 60 }
			any_other_country = { won_investor_candidate = yes has_country_flag = won_trade_@ROOT }
		}
		random_other_country = {
			limit = { won_investor_candidate = yes has_country_flag = won_trade_@ROOT }
			ROOT = { set_variable = { won_ev_partner = PREV } }
			set_variable = { global.won_ev_partner_name = THIS }
		}
		set_temp_variable = { won_found = 1 }
	}
	else_if = {
		limit = { any_other_country = { won_investor_candidate = yes } }
		random_other_country = {
			limit = { won_investor_candidate = yes }
			ROOT = { set_variable = { won_ev_partner = PREV } }
			set_variable = { global.won_ev_partner_name = THIS }
		}
		set_temp_variable = { won_found = 1 }
	}
}

# Fires the right offer for a human host, or settles it at once for an AI host.
won_investor_offer = {
	won_pick_investor = yes
	if = {
		limit = { check_variable = { won_found = 1 } }
		if = {
			limit = { is_ai = no }
			set_country_flag = { flag = won_fdi_offer_cd days = 90 }
			set_temp_variable = { won_qk = 3 }
			random_list = {
				60 = { set_temp_variable = { won_qk = 3 } }
				25 = {
					modifier = { factor = 0 NOT = { any_owned_state = { is_controlled_by = ROOT free_building_slots = { building = arms_factory size > 0 include_locked = no } } } }
					set_temp_variable = { won_qk = 4 }
				}
				15 = {
					modifier = { factor = 0 NOT = { any_owned_state = { is_controlled_by = ROOT is_coastal = yes free_building_slots = { building = dockyard size > 0 include_locked = no } } } }
					set_temp_variable = { won_qk = 5 }
				}
			}
			set_temp_variable = { won_qv = 0 }
			set_temp_variable = { won_qp = 1.5 }
			won_enqueue = yes
		}
		else = {
			set_country_flag = { flag = won_fdi_offer_cd days = 150 }
			won_accept_fdi_from_from = yes
		}
	}
}

won_fdi_inflow = {
	if = {
		limit = {
			check_variable = { won_climate > 0 }
			check_variable = { won_q_count < 5 }
			NOT = { has_country_flag = won_fdi_offer_cd }
			any_owned_state = {
				is_controlled_by = ROOT
				free_building_slots = { building = industrial_complex size > 0 include_locked = no }
			}
		}
		set_temp_variable_to_random = { var = won_roll min = 0 max = 100 }
		if = {
			limit = { check_variable = { won_roll < won_fdi_chance } }
			won_investor_offer = yes
		}
	}
}

# Reputation (0.8 to 1.2) times world tension (x1.1 above 25%, x1.2 above 50%). Stored in won_pf.
won_price_factor = {
	won_sync_reputation = yes
	set_variable = { won_pf = won_arms_rep }
	divide_variable = { won_pf = 250 }
	add_to_variable = { won_pf = 0.8 }
	if = {
		limit = { threat > 0.5 }
		multiply_variable = { won_pf = 1.2 }
	}
	else_if = {
		limit = { threat > 0.25 }
		multiply_variable = { won_pf = 1.1 }
	}
	set_variable = { won_pf_pct = won_pf }
	multiply_variable = { won_pf_pct = 100 }
}

		set_temp_variable = { won_delivered = 1 }

		limit = { check_variable = { won_k = 2 } has_equipment = { artillery_equipment > 124 } }

# Keeps contract k (temp vars) for next month.
won_keep_contract = {
	add_to_array = { won_tsw = won_p }
	add_to_array = { won_tsp = won_k }
	add_to_array = { won_tspr = won_pr }
	add_to_array = { won_tsl = won_left }
	add_to_array = { won_tsm = won_miss }
}

# Monthly, every country that sells: deliver, get paid, end, renew or break contracts.
won_supply_tick = {
	set_variable = { won_sc_income = 0 }
	clear_array = won_tsw
	clear_array = won_tsp
	clear_array = won_tspr
	clear_array = won_tsl
	clear_array = won_tsm
	for_each_loop = {
		array = won_sc_who
		value = won_p
		index = won_i
		set_temp_variable = { won_k = won_sc_pkg^won_i }
		set_temp_variable = { won_pr = won_sc_price^won_i }
		set_temp_variable = { won_left = won_sc_left^won_i }
		set_temp_variable = { won_miss = won_sc_miss^won_i }
		set_temp_variable = { won_ok = 1 }
		var:won_p = {
			if = {
				limit = {
					OR = {
						exists = no
						has_war_with = ROOT
						has_capitulated = yes
						has_country_flag = won_sc_cancel_@ROOT
					}
				}
				set_temp_variable = { won_ok = 0 }
			}
		}
		if = {
			limit = { check_variable = { won_ok = 1 } }
			won_try_delivery = yes
			if = {
				limit = { check_variable = { won_delivered = 1 } }
				var:won_p = { subtract_from_variable = { won_treasury = won_pr } }
				set_temp_variable = { won_spend = won_pr }
				won_arms_income = yes
				add_to_variable = { won_sc_income = won_pr }
				add_to_variable = { aon_rep = 6 }
				subtract_from_temp_variable = { won_left = 1 }
				set_temp_variable = { won_miss = 0 }
			}
			else = {
				add_to_temp_variable = { won_miss = 1 }
			}
			if = {
				limit = { check_variable = { won_miss > 1 } }
				# Two missed deliveries in a row: the buyer tears up the contract.
				var:won_p = {
					clr_country_flag = won_sc_from_@ROOT
					add_opinion_modifier = { target = ROOT modifier = won_contract_broken }
				}
				add_to_variable = { aon_rep = -40 }
				if = {
					limit = { is_ai = no }
					set_variable = { won_ev_partner = won_p }
					set_temp_variable = { won_qk = 7 }
					set_temp_variable = { won_qv = won_k }
					set_temp_variable = { won_qp = 0 }
					won_enqueue = yes
				}
			}
			else_if = {
				limit = { check_variable = { won_left < 1 } }
				if = {
					limit = { is_ai = no }
					# Completed: the renewal waits in the Foreign Trade Office (the buyer stays reserved).
					set_variable = { won_ev_partner = won_p }
					set_temp_variable = { won_qk = 6 }
					set_temp_variable = { won_qv = won_k }
					set_temp_variable = { won_qp = won_pr }
					won_enqueue = yes
				}
				else = {
					random_list = {
						50 = {
							set_temp_variable = { won_left = 6 }
							won_keep_contract = yes
						}
						50 = {
							var:won_p = { clr_country_flag = won_sc_from_@ROOT }
						}
					}
				}
			}
			else = {
				won_keep_contract = yes
			}
		}
		else = {
			var:won_p = {
				clr_country_flag = won_sc_from_@ROOT
				clr_country_flag = won_sc_cancel_@ROOT
			}
		}
	}
	clear_array = won_sc_who
	clear_array = won_sc_pkg
	clear_array = won_sc_price
	clear_array = won_sc_left
	clear_array = won_sc_miss
	set_variable = { won_sc_count = 0 }
	for_each_loop = {
		array = won_tsw
		value = won_p
		index = won_i
		add_to_array = { won_sc_who = won_p }
		set_temp_variable = { won_t = won_tsp^won_i }
		add_to_array = { won_sc_pkg = won_t }
		set_temp_variable = { won_t = won_tspr^won_i }
		add_to_array = { won_sc_price = won_t }
		set_temp_variable = { won_t = won_tsl^won_i }
		add_to_array = { won_sc_left = won_t }
		set_temp_variable = { won_t = won_tsm^won_i }
		add_to_array = { won_sc_miss = won_t }
		add_to_variable = { won_sc_count = 1 }
	}
	clear_array = won_tsw
	clear_array = won_tsp
	clear_array = won_tspr
	clear_array = won_tsl
	clear_array = won_tsm
}

# Royalties: 0.015 bn a month from every licensee still at peace with us, for 24 months.

	set_variable = { won_inc_roy = 0 }
	clear_array = won_tmp_who
	clear_array = won_tmp_amt
	for_each_loop = {
		array = won_roy_who
		value = won_p
		index = won_i
		set_temp_variable = { won_left = won_roy_left^won_i }
		subtract_from_temp_variable = { won_left = 1 }
		set_temp_variable = { won_ok = 1 }
		if = {
			limit = { check_variable = { won_left < 0 } }
			set_temp_variable = { won_ok = 0 }
		}
		var:won_p = {
			if = {
				limit = { OR = { exists = no has_war_with = ROOT } }
				set_temp_variable = { won_ok = 0 }
			}
		}
		if = {
			limit = { check_variable = { won_ok = 1 } }
			add_to_array = { won_tmp_who = won_p }
			add_to_array = { won_tmp_amt = won_left }
			add_to_variable = { won_inc_roy = 0.015 }
			var:won_p = { subtract_from_variable = { won_treasury = 0.015 } }
		}
	}
	clear_array = won_roy_who
	clear_array = won_roy_left
	for_each_loop = {
		array = won_tmp_who
		value = won_p
		index = won_i
		add_to_array = { won_roy_who = won_p }
		set_temp_variable = { won_left = won_tmp_amt^won_i }
		add_to_array = { won_roy_left = won_left }
	}
	clear_array = won_tmp_who
	clear_array = won_tmp_amt
	add_to_variable = { won_inc_special = won_inc_roy }
	add_to_variable = { won_acc_arms = won_inc_roy }
}

# Parallel arrays won_q_who / won_q_kind / won_q_val / won_q_price / won_q_age.
# Kinds: 1 supply contract offer, 2 licence request, 3 civilian factory investment,
#        4 arms plant investment, 5 shipyard investment, 6 contract up for renewal,
#        7 contract cancelled (notice). Requests wait 3 months, at most 5 at a time.

# Needs var won_ev_partner and temp vars won_qk (kind), won_qv (package/licence), won_qp (price).
won_enqueue = {
	add_to_array = { won_q_who = won_ev_partner }
	add_to_array = { won_q_kind = won_qk }
	add_to_array = { won_q_val = won_qv }
	add_to_array = { won_q_price = won_qp }
	add_to_array = { won_q_age = 0 }
	set_variable = { won_q_count = won_q_who^num }
	if = {
		limit = { is_ai = no }
		if = {
			limit = { has_country_flag = won_q_manual }
			if = {
				limit = { NOT = { has_country_flag = won_q_tutorial } }
				set_country_flag = won_q_tutorial
				country_event = { id = won.219 days = 1 }
			}
		}
		else = {
			# Pop-up mode (default): the request opens as an event right away, one at a time.
			set_country_flag = won_q_tutorial
			country_event = { id = won.221 hours = 2 }
		}
	}
}
